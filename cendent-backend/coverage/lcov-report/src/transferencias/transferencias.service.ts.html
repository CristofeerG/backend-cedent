export function generarCodigoTrz(): string {

  const fechaStr = hoy.toISOString().slice(0, 10).replace(/-/g, '');
  const aleatorio = Math.random().toString(36).substring(2, 7).toUpperCase();
  return `TRZ-${fechaStr}-${aleatorio}`;
}
